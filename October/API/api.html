Chapter 1: APIs Login Bypass

Playwright: login via API and open dashboard

import { test, expect, request, APIRequestContext } from "@playwright/test";

type LoginPayload = {
  email: string;
  password: string;
};

type LoginResponse = {
  token: string;
};

const payload: LoginPayload = {
  email: process.env.API_EMAIL ?? "",
  password: process.env.API_PASSWORD ?? "",
};

let token: string;
let apiContext: APIRequestContext;

test.beforeAll(async () => {
  apiContext = await request.newContext();
  const response = await apiContext.post(
    "https://bondaracademy.com/api/auth/sign-in/email",
    { data: payload },
  );
  await expect(response).toBeOK();
  const responseJson: LoginResponse = await response.json();
  token = responseJson.token;
});

test("login Bondar and click", async ({ page }) => {
  await page.addInitScript((value) => {
    window.localStorage.setItem("token", value);
  }, token);
  await page.goto("https://bondaracademy.com/dashboard");
});
