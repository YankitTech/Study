# Playwright + TypeScript — Large Website Test Suite

  ~240 test definitions (400+ executed cases once loops/viewports/browsers expand) covering
  functional E2E, API, accessibility, visual regression, responsive, security, SEO/site-health,
  network mocking, performance and UI-interaction patterns.

  Default target: https://www.saucedemo.com (public demo e-commerce site) so it runs out of the box.

  ## Quick start
  ```bash
  npm install
  npx playwright install --with-deps
  cp .env.example .env
  npm test                      # everything except visual
  npm run test:visual:update    # create visual baselines once
  npm run test:visual
  npm run report
  ```

  ## Structure
  ```
  playwright.config.ts          projects: setup, chromium, firefox, webkit, mobile-*, tablet, visual, api
  src/config/env.ts             env + budgets
  src/data/                     users, products, validation datasets
  src/pages/                    page objects (Base, Login, Inventory, ProductDetail, Cart, Checkout)
  src/fixtures/index.ts         custom fixtures (pages, pre-opened inventory, guestState)
  src/utils/helpers.ts          price parsing, sorting asserts, console/network health tracker
  tests/setup/                  logs in once -> .auth/standard-user.json (storageState)
  tests/auth/                   login, invalid-input matrix, session, access control, personas
  tests/inventory/              listing, sorting, product detail
  tests/cart/                   add/remove/persist/reset
  tests/checkout/               validation, price math, confirmation
  tests/e2e/                    full journeys
  tests/a11y/                   axe-core WCAG audits, keyboard, reduced-motion, forced-colors
  tests/visual/                 screenshot comparisons, dark mode, 3 viewports
  tests/responsive/             8 viewports x 4 checks, touch targets, orientation
  tests/security/               headers, cookies, XSS, CORS, TRACE, exposed files
  tests/seo/                    crawler-style health checks (CRAWL_PATHS), links, robots, sitemap
  tests/network/                route abort/fulfill/delay, HAR, offline, geolocation
  tests/performance/            nav timing, LCP/CLS, weight, CDP throttling
  tests/api/                    full REST CRUD against JSONPlaceholder
  tests/ui-interactions/        dialogs, frames, popups, upload/download, hover, tables, auth
  .github/workflows/            sharded matrix CI + merged HTML report
  ```

  ## Useful commands
  | Goal | Command |
  |---|---|
  | Smoke only | `npm run test:smoke` |
  | One browser | `npm run test:firefox` |
  | Mobile | `npm run test:mobile` |
  | API only | `npm run test:api` |
  | Accessibility | `npm run test:a11y` |
  | UI mode / debug | `npm run test:ui` / `npm run test:debug` |
  | Tag filter | `npx playwright test --grep "@security"` |

  Tags: `@smoke @regression @e2e @a11y @security @perf @api @responsive @slow`

  ## Pointing it at your own site
  1. Set `BASE_URL` (and `CRAWL_PATHS=/,/about,/pricing,...`) in `.env`.
  2. Replace selectors/data in `src/pages/*` and `src/data/*` (SauceDemo uses `data-test`, configured via `testIdAttribute`).
  3. Replace credentials in `.env`; the generic suites (seo, security, network, performance, a11y,
    responsive) work on any site with little change.

  ## Notes
  - Some checks are deliberately **annotated rather than failed** (security headers, serious a11y issues,
    SEO extras). Set `STRICT_SECURITY=true` to enforce headers. Annotations show in the HTML report.
  - Axe fails the build only on *critical* violations; full JSON is attached to each test.
  - The public demo sites can change or be slow; the `ui-interactions` and `api` suites depend on them.

src/pages/BasePage.ts

import { Locator, Page, expect } from '@playwright/test';

  export abstract class BasePage {
    abstract readonly path: string;

    constructor(readonly page: Page) {}

    // Header + sidebar are shared across every authenticated page
    get title(): Locator { return this.page.getByTestId('title'); }
    get cartLink(): Locator { return this.page.getByTestId('shopping-cart-link'); }
    get cartBadge(): Locator { return this.page.getByTestId('shopping-cart-badge'); }
    get menuButton(): Locator { return this.page.locator('#react-burger-menu-btn'); }
    get menuCloseButton(): Locator { return this.page.locator('#react-burger-cross-btn'); }
    get logoutLink(): Locator { return this.page.getByTestId('logout-sidebar-link'); }
    get resetLink(): Locator { return this.page.getByTestId('reset-sidebar-link'); }
    get aboutLink(): Locator { return this.page.getByTestId('about-sidebar-link'); }
    get allItemsLink(): Locator { return this.page.getByTestId('inventory-sidebar-link'); }

    async goto() {
      await this.page.goto(this.path);
    }

    async openMenu() {
      await this.menuButton.click();
      await expect(this.logoutLink).toBeVisible();
    }

    async logout() {
      await this.openMenu();
      await this.logoutLink.click();
      await expect(this.page).toHaveURL(/\/$/);
    }

    async resetAppState() {
      await this.openMenu();
      await this.resetLink.click();
      await this.menuCloseButton.click();
    }

    async openCart() {
      await this.cartLink.click();
      await expect(this.page).toHaveURL(/cart\.html/);
    }

    async cartCount(): Promise<number> {
      return (await this.cartBadge.count()) ? Number(await this.cartBadge.innerText()) : 0;
    }
  }

import { Locator, Page, expect } from '@playwright/test';
  import { BasePage } from './BasePage';
  import { Product } from '../data/products';
  import { parsePrice } from '../utils/helpers';

  export class InventoryPage extends BasePage {
    readonly path = '/inventory.html';

    readonly items: Locator;
    readonly sortSelect: Locator;

    constructor(page: Page) {
      super(page);
      this.items = page.getByTestId('inventory-item');
      this.sortSelect = page.getByTestId('product-sort-container');
    }

    async expectLoaded() {
      await expect(this.page).toHaveURL(/inventory\.html/);
      await expect(this.title).toHaveText('Products');
    }

    addButton(p: Product) { return this.page.getByTestId(`add-to-cart-${p.slug}`); }
    removeButton(p: Product) { return this.page.getByTestId(`remove-${p.slug}`); }
    itemCard(p: Product) { return this.items.filter({ hasText: p.name }); }

    async add(p: Product) { await this.addButton(p).click(); }
    async remove(p: Product) { await this.removeButton(p).click(); }

    async addMany(list: Product[]) { for (const p of list) await this.add(p); }

    async sortBy(value: 'az' | 'za' | 'lohi' | 'hilo') {
      await this.sortSelect.selectOption(value);
    }

    async names(): Promise<string[]> {
      return this.page.getByTestId('inventory-item-name').allInnerTexts();
    }

    async prices(): Promise<number[]> {
      return (await this.page.getByTestId('inventory-item-price').allInnerTexts()).map(parsePrice);
    }

    async imageSources(): Promise<string[]> {
      return this.items.locator('img').evaluateAll((imgs) => imgs.map((i) => (i as HTMLImageElement).src));

  /** Collects console errors, uncaught exceptions and failed requests for a page. */
  export function trackPageHealth(page: Page, opts: { ignore?: RegExp[] } = {}) {
    const consoleErrors: string[] = [];
    const pageErrors: string[] = [];
    const failedRequests: string[] = [];
    const ignore = opts.ignore ?? [];
    const skip = (s: string) => ignore.some((r) => r.test(s));

    page.on('console', (m) => {
      if (m.type() === 'error' && !skip(m.text())) consoleErrors.push(m.text());
    });
    page.on('pageerror', (e) => {
      if (!skip(e.message)) pageErrors.push(e.message);
    });
    page.on('requestfailed', (r) => {
      const s = `${r.method()} ${r.url()} - ${r.failure()?.errorText}`;
      if (!skip(s)) failedRequests.push(s);
    });
    return { consoleErrors, pageErrors, failedRequests };
  }
